payment_risk(Payment_amount, 10) :- Payment_amount < 1000.
payment_risk(Payment_amount, 60) :- Payment_amount >= 1000, Payment_amount =< 10000.
payment_risk(Payment_amount, 90) :- Payment_amount > 10000.

cardholder_risk(italy, 0).
cardholder_risk(Country, 10) :- continent(Country, C), C = europe.
cardholder_risk(Country, 20) :- continent(Country, C), C = africa.

merchant_risk(italy, 10).
merchant_risk(Country, 20) :- continent(Country, C), C = europe.
merchant_risk(Country, 60) :- continent(Country, C), C = africa.

additional_risk(swiss, south_africa, 30).
additional_risk(_, _, 0).

calculate_risk(Payment_amount, CardholderCountry, MerchantCountry, Total_risk) :-
    payment_risk(Payment_amount, PR), cardholder_risk(CardholderCountry, CR), 
    merchant_risk(MerchantCountry, MR), additional_risk(CardholderCountry, MerchantCountry, AR),
    Total_risk is PR + CR + MR + AR, !.

check_time(Transaction, CardID, Datetime, M_Country, physical, TR, 150) 
		:- transaction_cardId(T, CardID), transaction_datetime(T, DT),
    		transaction_merchant(T, M), merchant_country(M, M_CO),  store_type(M, physical),
    		Transaction \== T, M_Country \== M_CO, 
    		Diff is Datetime - DT, Diff >= 10, Diff =< 300, TR < 150, !. 
check_time(_, _, _, _, _, TR, TR).	


check_risk(Risk, "no_risk") :- Risk =< 100, !.
check_risk(Risk, "transaction stopped") :- Risk > 100, Risk =< 150, !.
check_risk(Risk, "transaction declined") :- Risk > 150.	


check_transaction(Transaction, Result) :- payment_amount(Transaction, P), transaction_cardId(Transaction, Card_ID), 
					  transaction_datetime(Transaction, Datetime),  cardholder(Card_ID, C), 
					  cardholder_country(C, C_CO), transaction_merchant(Transaction, M),
    				 merchant_country(M, M_CO), store_type(M, S_T), calculate_risk(P, C_CO, M_CO, TR), 
      					check_time(Transaction, Card_ID, Datetime,M_CO, S_T, TR, Adjusted_Risk), 
					  check_risk(Adjusted_Risk, Result).
